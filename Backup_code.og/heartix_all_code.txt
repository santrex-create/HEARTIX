/*
 * ============================================================================
 * HEARTIX â€” REAL-TIME BIOMEDICAL ECG TELEMETRY FIRMWARE (ESP32 + AD8232)
 * ============================================================================
 *
 * ****************************************************************************
 * *                     CRITICAL ELECTRICAL SAFETY WARNING                   *
 * ****************************************************************************
 * * NEVER connect the ESP32 to a mains-powered USB wall charger, desktop PC, *
 * * or plugged-in laptop while AD8232 electrodes are attached to a human     *
 * * body! Mains leakage currents can cause fatal ventricular fibrillation.   *
 * *                                                                          *
 * * ALWAYS power the ESP32 exclusively from an isolated USB power bank or    *
 * * LiPo battery during human ECG acquisition, and use an unplugged laptop   *
 * * running on battery if viewing the Serial Monitor.                        *
 * ****************************************************************************
 *
 * Architecture:
 *  - Core 1 (Priority 3): Deterministic 250 Hz ADC1 sampling via vTaskDelayUntil
 *  - Core 0 (Priority 2): WiFiMulti + WSS (TLS) streaming via links2004/WebSockets
 *  - Zero-copy lock-free double buffering (54-byte packed binary frames, 10 Hz)
 */

#include <Arduino.h>
#include <WiFi.h>
#include <WiFiMulti.h>
#include <WebSocketsClient.h>

// ============================================================================
// USER CONFIGURATION â€” UPDATE THESE 3 SECTIONS BEFORE FLASHING
// ============================================================================

// 1. 2.4 GHz Wi-Fi Credentials (ESP32 does not support 5 GHz Wi-Fi)
#define WIFI_SSID_1      "YOUR_PRIMARY_WIFI_SSID"
#define WIFI_PASS_1      "YOUR_PRIMARY_WIFI_PASSWORD"

#define WIFI_SSID_2      "YOUR_MOBILE_HOTSPOT_SSID"
#define WIFI_PASS_2      "YOUR_MOBILE_HOTSPOT_PASSWORD"

// 2. Render Cloud Hostname (DO NOT include "https://" or trailing "/")
#define RENDER_HOST      "heartix-monitor.onrender.com"

// 3. Device Authentication Token (MUST match HEARTIX_DEVICE_TOKEN on Render)
#define DEVICE_TOKEN     "replace-with-your-secret-heartix-token"

// ============================================================================
// HARDWARE PIN DEFINITIONS (STRICTLY ADC1 TO AVOID WI-FI DRIVER CONFLICTS)
// ============================================================================
constexpr uint8_t ECG_PIN        = 34; // ADC1_CH6 (Input-only, clean analog path)
constexpr uint8_t LO_PLUS_PIN    = 32; // AD8232 LO+ Leads-Off Digital Output
constexpr uint8_t LO_MINUS_PIN   = 33; // AD8232 LO- Leads-Off Digital Output
constexpr uint8_t STATUS_LED_PIN = 2;  // Onboard ESP32 Blue LED

// ============================================================================
// LOCKED BINARY PROTOCOL CONTRACT (54 BYTES LITTLE-ENDIAN, MATCHING main.py)
// ============================================================================
constexpr uint16_t SAMPLE_RATE_HZ     = 250;
constexpr TickType_t SAMPLE_PERIOD_MS = 4;   // 1000 ms / 250 Hz = 4 ms exact
constexpr uint8_t  BATCH_SIZE         = 25;  // 25 samples * 4 ms = 100 ms (10 pkt/s)

struct __attribute__((packed)) EcgBinaryPacket {
    uint16_t seq;                  // Bytes 0-1: incrementing sequence counter (0..65535)
    uint8_t  flags;                // Byte 2: bit 0 = leads_off (1 if LO+ or LO- HIGH)
    uint8_t  count;                // Byte 3: always 25
    uint16_t samples[BATCH_SIZE];  // Bytes 4-53: calibrated ADC readings in mV (0..3300)
};

static_assert(sizeof(EcgBinaryPacket) == 54, "Packet size must be 54 bytes");

// ============================================================================
// DOUBLE-BUFFER & SINGLE-QUEUE IPC STATE (NO MUTEXES REQUIRED)
// ============================================================================
static EcgBinaryPacket txBuffers[2];
static QueueHandle_t   readyBufferQueue = nullptr;

// Shared state flags between Core 0 and Core 1 (atomic single-byte reads/writes)
static volatile bool wsConnected = false;

// ============================================================================
// CORE 1 TASK: DETERMINISTIC 250 HZ ECG SAMPLING
// ============================================================================
void samplingTask(void* pvParameters) {
    (void)pvParameters;

    // Configure ADC1 for 0 - 3300 mV calibrated millivolt output
    analogReadResolution(12);
    analogSetPinAttenuation(ECG_PIN, ADC_11db);

    uint8_t  activeBufIdx = 0;
    uint8_t  sampleIdx    = 0;
    uint16_t packetSeq    = 0;
    bool     batchLeadsOff = false;

    TickType_t lastWakeTime = xTaskGetTickCount();

    for (;;) {
        // 1. Read raw calibrated millivolts (NO moving average / EMA filtering)
        uint16_t mv = static_cast<uint16_t>(analogReadMilliVolts(ECG_PIN));

        // 2. Check hardware leads-off comparator pins (HIGH = electrode detached)
        if (digitalRead(LO_PLUS_PIN) == HIGH || digitalRead(LO_MINUS_PIN) == HIGH) {
            batchLeadsOff = true;
        }

        // 3. Store into active double-buffer slot
        txBuffers[activeBufIdx].samples[sampleIdx] = mv;
        sampleIdx++;

        // 4. Once 25 samples (100 ms) are collected, seal packet and hand off to Core 0
        if (sampleIdx >= BATCH_SIZE) {
            txBuffers[activeBufIdx].seq   = packetSeq++;
            txBuffers[activeBufIdx].flags = batchLeadsOff ? 0x01 : 0x00;
            txBuffers[activeBufIdx].count = BATCH_SIZE;

            uint8_t completedIdx = activeBufIdx;

            // Swap to the alternate buffer immediately for the next 4 ms tick
            activeBufIdx ^= 1;
            sampleIdx     = 0;
            batchLeadsOff = false;

            // Non-blocking queue send: if Core 0 is stalled on Wi-Fi reconnect,
            // drop oldest/skip without ever delaying Core 1's 250 Hz clock.
            xQueueSend(readyBufferQueue, &completedIdx, 0);
        }

        // 5. Sleep until the exact next 4 ms boundary
        vTaskDelayUntil(&lastWakeTime, pdMS_TO_TICKS(SAMPLE_PERIOD_MS));
    }
}

// ============================================================================
// CORE 0 WEBSOCKET CLIENT & EVENT HANDLER
// ============================================================================
static WiFiMulti        wifiMulti;
static WebSocketsClient webSocket;

void onWebSocketEvent(WStype_t type, uint8_t* payload, size_t length) {
    (void)length;
    switch (type) {
        case WStype_DISCONNECTED:
            if (wsConnected) {
                Serial.println("[WS] Disconnected from Render backend. Reconnecting...");
            }
            wsConnected = false;
            break;

        case WStype_CONNECTED:
            wsConnected = true;
            Serial.printf("[WS] Connected to wss://%s%s\n", RENDER_HOST, (char*)payload);
            break;

        case WStype_ERROR:
            wsConnected = false;
            Serial.println("[WS] TLS/WebSocket error encountered.");
            break;

        case WStype_TEXT:
        case WStype_BIN:
        case WStype_PING:
        case WStype_PONG:
        default:
            break;
    }
}

// ============================================================================
// STATUS LED CONTROLLER (NON-BLOCKING)
//  - Slow blink (500 ms) : Connecting to Wi-Fi
//  - Fast blink (125 ms) : Wi-Fi OK, connecting to Render WSS
//  - Solid ON            : Live streaming ECG packets
// ============================================================================
void updateStatusLed(bool wifiReady, bool serverReady) {
    static uint32_t lastToggleMs = 0;
    static bool     ledState     = false;

    if (wifiReady && serverReady) {
        if (!ledState) {
            ledState = true;
            digitalWrite(STATUS_LED_PIN, HIGH);
        }
        return;
    }

    uint32_t intervalMs = wifiReady ? 125 : 500;
    uint32_t now        = millis();

    if (now - lastToggleMs >= intervalMs) {
        lastToggleMs = now;
        ledState     = !ledState;
        digitalWrite(STATUS_LED_PIN, ledState ? HIGH : LOW);
    }
}

// ============================================================================
// CORE 0 TASK: WI-FI, TLS WEBSOCKET & PACKET TRANSMISSION
// ============================================================================
void networkTask(void* pvParameters) {
    (void)pvParameters;

    WiFi.mode(WIFI_STA);
    WiFi.setSleep(false); // Disable modem sleep for low-latency 100 ms packet cadence

    wifiMulti.addAP(WIFI_SSID_1, WIFI_PASS_1);
    if (strlen(WIFI_SSID_2) > 0 && strcmp(WIFI_SSID_2, "YOUR_MOBILE_HOTSPOT_SSID") != 0) {
        wifiMulti.addAP(WIFI_SSID_2, WIFI_PASS_2);
    }

    Serial.println("[WIFI] Connecting to 2.4 GHz access point...");
    while (wifiMulti.run() != WL_CONNECTED) {
        updateStatusLed(false, false);
        vTaskDelay(pdMS_TO_TICKS(100));
    }

    Serial.printf("[WIFI] Connected! IP: %s | RSSI: %d dBm\n",
                  WiFi.localIP().toString().c_str(), WiFi.RSSI());

    // Build authenticated WebSocket path: /ws/esp32?token=<HEARTIX_DEVICE_TOKEN>
    String wsPath = "/ws/esp32?token=";
    wsPath += DEVICE_TOKEN;

    // Configure WSS (TLS on port 443)
    // Note: beginSSL without root CA fingerprint enables standard TLS encryption
    // compatible with Render's auto-renewing Let's Encrypt certificates.
    webSocket.beginSSL(RENDER_HOST, 443, wsPath.c_str());
    webSocket.onEvent(onWebSocketEvent);
    webSocket.setReconnectInterval(3000);
    webSocket.enableHeartbeat(15000, 3000, 2);

    uint32_t packetsSentWindow  = 0;
    uint32_t packetsDropWindow  = 0;
    uint32_t lastDiagPrintMs    = millis();
    bool     wasWifiConnected   = true;

    for (;;) {
        bool wifiOk = (wifiMulti.run() == WL_CONNECTED);

        if (wifiOk && !wasWifiConnected) {
            Serial.printf("[WIFI] Reconnected! IP: %s\n", WiFi.localIP().toString().c_str());
        } else if (!wifiOk && wasWifiConnected) {
            Serial.println("[WIFI] Connection lost. Searching for AP...");
            wsConnected = false;
        }
        wasWifiConnected = wifiOk;

        if (wifiOk) {
            webSocket.loop();
        }

        // Wait up to 10 ms for a completed 25-sample buffer from Core 1
        uint8_t readyIdx = 0;
        if (xQueueReceive(readyBufferQueue, &readyIdx, pdMS_TO_TICKS(10)) == pdTRUE) {
            if (wifiOk && wsConnected) {
                bool ok = webSocket.sendBIN(
                    reinterpret_cast<uint8_t*>(&txBuffers[readyIdx]),
                    sizeof(EcgBinaryPacket)
                );
                if (ok) {
                    packetsSentWindow++;
                } else {
                    packetsDropWindow++;
                }
            } else {
                packetsDropWindow++;
            }
        }

        updateStatusLed(wifiOk, wsConnected);

        // Clean 5-second diagnostic telemetry summary (never spams Serial at 250 Hz)
        uint32_t now = millis();
        if (now - lastDiagPrintMs >= 5000) {
            float pktRate = packetsSentWindow / 5.0f;
            Serial.printf(
                "[TELEMETRY] WiFi=%s | WS=%s | Rate=%.1f pkt/s (target 10.0) | Sent=%lu | Dropped=%lu | Heap=%lu B\n",
                wifiOk ? "OK" : "DOWN",
                wsConnected ? "LIVE" : "WAIT",
                pktRate,
                static_cast<unsigned long>(packetsSentWindow),
                static_cast<unsigned long>(packetsDropWindow),
                static_cast<unsigned long>(ESP.getFreeHeap())
            );
            packetsSentWindow = 0;
            packetsDropWindow = 0;
            lastDiagPrintMs   = now;
        }
    }
}

// ============================================================================
// ARDUINO SETUP & LOOP
// ============================================================================
void setup() {
    Serial.begin(115200);
    delay(200);

    Serial.println("\n====================================================");
    Serial.println("  HEARTIX BIOMEDICAL TELEMETRY NODE (ESP32-WROOM)   ");
    Serial.println("  Protocol: 54B Packed Binary @ 250 Hz (10 pkt/s)   ");
    Serial.println("====================================================");

    pinMode(LO_PLUS_PIN, INPUT);
    pinMode(LO_MINUS_PIN, INPUT);
    pinMode(STATUS_LED_PIN, OUTPUT);
    digitalWrite(STATUS_LED_PIN, LOW);

    memset(txBuffers, 0, sizeof(txBuffers));

    // Single 2-slot queue passing buffer index (0 or 1) from Core 1 -> Core 0
    readyBufferQueue = xQueueCreate(2, sizeof(uint8_t));
    if (readyBufferQueue == nullptr) {
        Serial.println("[FATAL] Failed to create FreeRTOS queue. Halting.");
        while (true) {
            delay(1000);
        }
    }

    // Pin deterministic 250 Hz ADC sampling to Core 1 (Application Core)
    xTaskCreatePinnedToCore(
        samplingTask,
        "EcgSamplingTask",
        4096,
        nullptr,
        3,    // High priority for strict 4 ms timing
        nullptr,
        1     // Core 1
    );

    // Pin Wi-Fi + TLS WebSocket stack to Core 0 (Protocol Core)
    xTaskCreatePinnedToCore(
        networkTask,
        "EcgNetworkTask",
        8192, // Generous stack for mbedTLS / WSS handshake
        nullptr,
        2,    // Priority 2
        nullptr,
        0     // Core 0
    );
}

void loop() {
    // All work runs inside pinned FreeRTOS tasks; delete default loopTask to free RAM
    vTaskDelete(nullptr);
}

        evaluateSystemState();
        break;

        if (state.monitoringStopped) return;
        state.leadsOff = Boolean(msg.leads_off);
        state.calibrating = Boolean(msg.calibrating);
        
        if (msg.source && DOM.metaSource) {
          DOM.metaSource.textContent = `SOURCE: ${String(msg.source).toUpperCase()}`;

          }
        }
        evaluateSystemState();
        break;

      case "metrics":
        updateMetricsDisplay(msg);
        break;

      case "reset":
        resetEcgBuffers();
        break;

      default:
        break;
    }

    if (DOM.valAiReason && m.ai_reason) {
      DOM.valAiReason.textContent = m.ai_reason;
    }

    if (DOM.flagBrady) {
      DOM.flagBrady.className = m.bradycardia ? "rhythm-flag flag-active-amber" : "rhythm-flag";

    if (DOM.eventCountBadge && m.event_count !== undefined) {
      DOM.eventCountBadge.textContent = m.event_count;

    // Audio Alarms
    if (m.alarm_level && window.heartixAudio) {

      updateAlarmBanner(m.alarm_level, m.status || "CLINICAL ALARM ACTIVE", m.ai_reason || "Physiological abnormality detected");

  function resizeCanvas() {
    if (!DOM.ecgCanvas) return;
    const parent = DOM.ecgCanvas.parentElement;
    if (!parent) return;

    const dpr = window.devicePixelRatio || 1;
    const width = parent.clientWidth;
    const height = parent.clientHeight;

    DOM.ecgCanvas.width = width * dpr;
    DOM.ecgCanvas.height = height * dpr;
    DOM.ecgCanvas.style.width = width + "px";
    DOM.ecgCanvas.style.height = height + "px";

    const ctx = DOM.ecgCanvas.getContext("2d");
    ctx.scale(dpr, dpr);
  }

  function renderLoop() {
    drawEcgCanvas();
    requestAnimationFrame(renderLoop);
  }

  function drawEcgCanvas() {
    const canvas = DOM.ecgCanvas;
    if (!canvas) return;
    const ctx = canvas.getContext("2d");
    const dpr = window.devicePixelRatio || 1;
    const width = canvas.width / dpr;
    const height = canvas.height / dpr;

    // 1. Clear background & draw ECG grid
    ctx.fillStyle = "#04070B";
    ctx.fillRect(0, 0, width, height);

    drawEcgGrid(ctx, width, height);

    // 2. Process incoming sample queue into buffer
    if (!state.frozen && !state.monitoringStopped && state.sampleQueue.length > 0) {
      // Consume up to 8 samples per frame to maintain real-time sync without backlog lag
      const batchSize = Math.min(state.sampleQueue.length, 8);
      for (let i = 0; i < batchSize; i++) {
        const item = state.sampleQueue.shift();
        state.ecgBuffer.push(item);
        state.rollingPrintBuffer.push(item);
        if (state.rollingPrintBuffer.length > 2500) {
          state.rollingPrintBuffer.shift();
        }
      }
    }

    const maxPoints = state.sweepSpeedMmPerSec === 50 ? 625 : 1250; // 250 Hz * 2.5s or 5.0s
    if (state.ecgBuffer.length > maxPoints) {
      state.ecgBuffer = state.ecgBuffer.slice(state.ecgBuffer.length - maxPoints);
    }

    if (state.ecgBuffer.length < 2) return;

    // 3. Determine vertical scaling (Gain)
    let scaleY = height * 0.18; // default 1x
    if (state.gainMode === "2X") scaleY *= 2;
    if (state.gainMode === "AUTO") {
      // Auto-scale based on peak amplitudes in buffer
      let maxAbs = 1.0;
      for (let p of state.ecgBuffer) {
        if (Math.abs(p.sample) > maxAbs) maxAbs = Math.abs(p.sample);
      }
      scaleY = (height * 0.38) / Math.max(maxAbs, 0.5);
    }

    const centerY = height / 2;
    const dx = width / maxPoints;

    // 4. Draw sweeping ECG trace
    ctx.lineWidth = 2.0;
    ctx.strokeStyle = "#00E676"; // Hospital Green
    ctx.shadowColor = "rgba(0, 230, 118, 0.55)";
    ctx.shadowBlur = 6;

    ctx.beginPath();
    for (let i = 0; i < state.ecgBuffer.length; i++) {
      const pt = state.ecgBuffer[i];
      const x = i * dx;
      const y = centerY - (pt.sample * scaleY);

      if (i === 0) ctx.moveTo(x, y);
      else ctx.lineTo(x, y);

      // Trigger QRS beep & heart pulse if R-peak is encountered near write head
      if (pt.rPeak && !state.frozen && state.validSignal && i === state.ecgBuffer.length - 1) {
        triggerHeartBeatFx();
      }
    }
    ctx.stroke();
    ctx.shadowBlur = 0; // reset shadow

    // 5. Draw R-peak markers (small cyan/red triangles above R peaks)
    ctx.fillStyle = "#38BDF8";
    for (let i = 0; i < state.ecgBuffer.length; i++) {
      const pt = state.ecgBuffer[i];
      if (pt.rPeak) {
        const x = i * dx;
        const y = centerY - (pt.sample * scaleY) - 14;
        ctx.beginPath();
        ctx.moveTo(x, y);
        ctx.lineTo(x - 4, y - 6);
        ctx.lineTo(x + 4, y - 6);
        ctx.closePath();
        ctx.fill();
      }
    }

    // 6. Draw moving eraser/sweep head bar if not frozen
    if (!state.frozen && !state.monitoringStopped) {
      const headX = (state.ecgBuffer.length - 1) * dx;
      ctx.fillStyle = "rgba(0, 230, 118, 0.35)";
      ctx.fillRect(headX, 0, 4, height);
    }
  }

  function drawEcgGrid(ctx, width, height) {
    const minorSize = 5; // mm approx pixels
    const majorSize = 25;

    // Minor grid lines
    ctx.strokeStyle = "rgba(10, 30, 20, 0.4)";
    ctx.lineWidth = 0.5;
    ctx.beginPath();
    for (let x = 0; x < width; x += minorSize) {
      ctx.moveTo(x, 0); ctx.lineTo(x, height);
    }
    for (let y = 0; y < height; y += minorSize) {
      ctx.moveTo(0, y); ctx.lineTo(width, y);
    }
    ctx.stroke();

    // Major grid lines
    ctx.strokeStyle = "rgba(18, 60, 40, 0.65)";
    ctx.lineWidth = 1;
    ctx.beginPath();
    for (let x = 0; x < width; x += majorSize) {
      ctx.moveTo(x, 0); ctx.lineTo(x, height);
    }
    for (let y = 0; y < height; y += majorSize) {
      ctx.moveTo(0, y); ctx.lineTo(width, y);
    }
    ctx.stroke();
  }

  function triggerHeartBeatFx() {
    if (window.heartixAudio) {
      window.heartixAudio.playQrsBeep();
    }
    if (DOM.heartIcon) {
      DOM.heartIcon.classList.add("beat-pulse");
      setTimeout(() => {
        if (DOM.heartIcon) DOM.heartIcon.classList.remove("beat-pulse");
      }, 90);
    }
  }

  async function submitPatientAdmission(isGuest) {
    if (window.heartixAudio) {
      window.heartixAudio.unlock();
    }

    let patientPayload = {
      name: state.patient.name,
      age: Number(state.patient.age) || 45,
      gender: state.patient.gender,
      patient_id: state.patient.id,
      bed: state.patient.bed,
      min_hr: Number(state.patient.minHr) || 60,
      max_hr: Number(state.patient.maxHr) || 100,
      notes: state.patient.notes
    };

    if (!isGuest && DOM.patientForm) {
      patientPayload = {
        name: DOM.inpPatientName.value || "Aarav Sharma",
        age: Number(DOM.inpPatientAge.value) || 45,
        gender: DOM.inpPatientGender.value || "Male",
        patient_id: DOM.inpPatientId.value || "HTX-2026-001",
        bed: DOM.inpPatientBed.value || "ICU-01",
        min_hr: Number(DOM.inpMinHr.value) || 60,
        max_hr: Number(DOM.inpMaxHr.value) || 100,
        notes: DOM.inpNotes.value || ""
      };
    }

    try {
      const res = await fetch("/api/patient", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify(patientPayload)
      });
      if (res.ok) {
        const data = await res.json();
        syncPatientData(data.patient || patientPayload);
      }
    } catch (err) {
      console.error("Failed to submit patient admission:", err);
      syncPatientData(patientPayload);
    }

    if (DOM.patientModal) {
      DOM.patientModal.classList.add("hidden");
    }
  }

  function renderEventsList(events) {
    if (!DOM.eventsList) return;
    if (!events || events.length === 0) {
      DOM.eventsList.innerHTML = `<div class="empty-events">No clinical arrhythmia or threshold events recorded in current session.</div>`;
      return;
    }

    DOM.eventsList.innerHTML = events.map((ev, idx) => `
      <div class="event-item ${ev.severity === 'high' ? 'severity-high' : ''}" data-index="${idx}">
        <div style="display:flex; justify-content:space-between; font-weight:700; font-size:12px; margin-bottom:3px;">
          <span style="color:var(--text-primary);">${ev.type || 'CLINICAL EVENT'}</span>
          <span style="font-family:var(--font-mono); font-size:10px; color:var(--text-secondary);">${ev.timestamp || '--:--:--'}</span>
        </div>
        <div style="font-size:11.5px; color:var(--text-secondary);">${ev.description || ev.reason || 'Telemetry threshold breach'}</div>
      </div>
    `).join("");

    const itemEls = DOM.eventsList.querySelectorAll(".event-item");
    itemEls.forEach(el => {
      el.addEventListener("click", (e) => {
        itemEls.forEach(i => i.classList.remove("selected"));
        e.currentTarget.classList.add("selected");
        const idx = Number(e.currentTarget.getAttribute("data-index"));
        renderEventSnapshot(events[idx]);
      });
    });

    // Auto-select first event if available
    if (events.length > 0) {
      itemEls[0].classList.add("selected");
      renderEventSnapshot(events[0]);
    }
  }

  function renderEventSnapshot(ev) {
    if (DOM.eventSnapshotMeta) {
      DOM.eventSnapshotMeta.textContent = `${ev.type} @ ${ev.timestamp} (${ev.severity || 'medium'} severity)`;
    }
    const canvas = DOM.eventSnapshotCanvas;
    if (!canvas) return;
    const ctx = canvas.getContext("2d");
    const width = canvas.clientWidth || 380;
    const height = canvas.clientHeight || 120;
    canvas.width = width * (window.devicePixelRatio || 1);
    canvas.height = height * (window.devicePixelRatio || 1);
    ctx.scale(window.devicePixelRatio || 1, window.devicePixelRatio || 1);

    ctx.fillStyle = "#04070B";
    ctx.fillRect(0, 0, width, height);

    // Draw grid
    ctx.strokeStyle = "rgba(10, 30, 20, 0.5)";
    ctx.lineWidth = 0.5;
    for (let x = 0; x < width; x += 20) {
      ctx.moveTo(x, 0); ctx.lineTo(x, height);
    }
    for (let y = 0; y < height; y += 20) {
      ctx.moveTo(0, y); ctx.lineTo(width, y);
    }
    ctx.stroke();

    // Plot waveform snapshot samples if present in event, else sample rolling buffer
    const samples = (ev.waveform_strip && ev.waveform_strip.length > 0) 
      ? ev.waveform_strip 
      : state.rollingPrintBuffer.map(p => p.sample).slice(-300);

    if (!samples || samples.length === 0) return;

    const dx = width / samples.length;
    const centerY = height / 2;
    ctx.strokeStyle = "#38BDF8";
    ctx.lineWidth = 1.8;
    ctx.beginPath();
    for (let i = 0; i < samples.length; i++) {
      const val = typeof samples[i] === 'object' ? samples[i].sample : samples[i];
      const x = i * dx;
      const y = centerY - (val * (height * 0.28));
      if (i === 0) ctx.moveTo(x, y);
      else ctx.lineTo(x, y);
    }
    ctx.stroke();
  }

  // --- PDF REPORT & PRINT STYLESHEET GENERATOR ---
  function generatePrintReport() {
    if (DOM.printTimestamp) {
      DOM.printTimestamp.textContent = new Date().toLocaleString();
    }

    if (DOM.printPatientInfo) {
      DOM.printPatientInfo.innerHTML = `
        <div><strong>Patient Name:</strong> ${state.patient.name}</div>
        <div><strong>Patient ID:</strong> ${state.patient.id}</div>
        <div><strong>Age / Gender:</strong> ${state.patient.age} yrs / ${state.patient.gender}</div>
        <div><strong>Bed / Ward:</strong> ${state.patient.bed}</div>
      `;
    }

    // Render static pink-grid ECG strip canvas for print
    const printCanvas = DOM.printEcgCanvas;
    if (printCanvas) {
      const width = printCanvas.clientWidth || 900;
      const height = printCanvas.clientHeight || 190;
      printCanvas.width = width * (window.devicePixelRatio || 1);
      printCanvas.height = height * (window.devicePixelRatio || 1);
      const ctx = printCanvas.getContext("2d");
      ctx.scale(window.devicePixelRatio || 1, window.devicePixelRatio || 1);

      ctx.fillStyle = "#FFF5F5";
      ctx.fillRect(0, 0, width, height);

      // Pink ECG grid lines
      ctx.strokeStyle = "rgba(225, 29, 72, 0.22)";
      ctx.lineWidth = 0.6;
      ctx.beginPath();
      for (let x = 0; x < width; x += 25) {
        ctx.moveTo(x, 0); ctx.lineTo(x, height);
      }
      for (let y = 0; y < height; y += 25) {
        ctx.moveTo(0, y); ctx.lineTo(width, y);
      }
      ctx.stroke();

      const printSamples = state.rollingPrintBuffer.length > 0 
        ? state.rollingPrintBuffer 
        : state.ecgBuffer;

/**
 * HEARTIX â€” Web Audio API Clinical Telemetry & IEC 60601-1-8 Inspired Alarm Engine
 * Zero external audio assets; pure synthesized sine tones with click-free ADSR envelopes.
 */
class HeartixAudioEngine {
  constructor() {
    this.ctx = null;
    this.unlocked = false;
    this.qrsBeepEnabled = true;
    this.alarmLevel = "none";

    // Repeating alarm scheduler state
    this._alarmIntervalId = null;
    this._scheduledTimeoutIds = [];

    // 2-minute ICU audio pause state
    this._muted = false;
    this._muteRemaining = 0;
    this._muteTimerId = null;
    this._onMuteTick = null;
  }

  /**
   * Creates or resumes AudioContext cleanly on a user gesture.
   */
  unlock() {
    if (!this.ctx) {
      const AudioCtx = window.AudioContext || window.webkitAudioContext;
      if (AudioCtx) {
        this.ctx = new AudioCtx();
      }
    }
    if (this.ctx && this.ctx.state === "suspended") {
      this.ctx.resume().catch(() => {});
    }
    if (this.ctx) {
      this.unlocked = true;
    }
    return this.unlocked;
  }

  /**
   * Internal helper to synthesize a single click-free sine tone burst.
   */
  _playTone(freqHz, durationSec, peakGain = 0.14, delaySec = 0) {
    if (!this.unlocked || !this.ctx) return;
    if (this.ctx.state === "suspended") {
      this.ctx.resume().catch(() => {});
    }

    const now = this.ctx.currentTime + delaySec;
    const attack = 0.005;  // 5 ms smooth attack
    const release = 0.005; // 5 ms smooth release
    const sustainEnd = Math.max(now + attack, now + durationSec - release);

    const osc = this.ctx.createOscillator();
    const gainNode = this.ctx.createGain();

    osc.type = "sine";
    osc.frequency.setValueAtTime(freqHz, now);

    gainNode.gain.setValueAtTime(0.0001, now);
    gainNode.gain.exponentialRampToValueAtTime(Math.max(0.0002, peakGain), now + attack);
    gainNode.gain.setValueAtTime(Math.max(0.0002, peakGain), sustainEnd);
    gainNode.gain.exponentialRampToValueAtTime(0.0001, now + durationSec);

    osc.connect(gainNode);
    gainNode.connect(this.ctx.destination);

    osc.start(now);
    osc.stop(now + durationSec + 0.01);
  }

  /**
   * Plays a crisp 880 Hz, 55 ms R-peak QRS tone with 5 ms attack/release.
   */
  playQrsBeep() {
    if (!this.unlocked || !this.qrsBeepEnabled || this._muted) return;
    this._playTone(880, 0.055, 0.11, 0);
  }

  /**
   * Enables or disables the R-peak QRS systolic tone.
   * @param {boolean} enabled
   */
  setQrsBeepEnabled(enabled) {
    this.qrsBeepEnabled = Boolean(enabled);
  }

  /**
   * Clears any active alarm intervals and pending burst pulse timeouts.
   */
  _clearAlarmTimers() {
    if (this._alarmIntervalId !== null) {
      clearInterval(this._alarmIntervalId);
      this._alarmIntervalId = null;
    }
    while (this._scheduledTimeoutIds.length > 0) {
      clearTimeout(this._scheduledTimeoutIds.pop());
    }
  }

  /**
   * Schedules an urgent 5-pulse High Priority burst: (beep-beep-beep â€” beep-beep) at 960 Hz.
   */
  _emitHighBurst() {
    if (this._muted || !this.unlocked) return;
    // Pattern offsets in ms: 3 rapid pulses, brief pause, 2 rapid pulses
    const offsetsMs = [0, 140, 280, 520, 660];
    offsetsMs.forEach((offset) => {
      const id = setTimeout(() => {
        if (!this._muted && this.alarmLevel === "high") {
          this._playTone(960, 0.085, 0.18, 0);
        }
      }, offset);
      this._scheduledTimeoutIds.push(id);
    });
  }

  /**
   * Schedules a 3-pulse Medium Priority warning burst: (beep â€” beep â€” beep) at 660 Hz.
   */
  _emitMediumBurst() {
    if (this._muted || !this.unlocked) return;
    const offsetsMs = [0, 240, 480];
    offsetsMs.forEach((offset) => {
      const id = setTimeout(() => {
        if (!this._muted && this.alarmLevel === "medium") {
          this._playTone(660, 0.11, 0.15, 0);
        }
      }, offset);
      this._scheduledTimeoutIds.push(id);
    });
  }

  /**
   * Sets clinical alarm priority level and manages IEC 60601-1-8 inspired repeating bursts.
   * @param {"none" | "technical" | "low" | "medium" | "high"} level
   */
  setAlarmLevel(level) {
    const normalized = String(level || "none").toLowerCase();
    if (this.alarmLevel === normalized && (this._alarmIntervalId !== null || (normalized !== "high" && normalized !== "medium"))) {
      return;
    }

    this.alarmLevel = normalized;
    this._clearAlarmTimers();

    if (normalized === "high") {
      this._emitHighBurst();
      this._alarmIntervalId = setInterval(() => {
        this._emitHighBurst();
      }, 5000);
    } else if (normalized === "medium") {
      this._emitMediumBurst();
      this._alarmIntervalId = setInterval(() => {
        this._emitMediumBurst();
      }, 8000);
    }
    // "none", "technical", "low" stop repeating clinical alarms
  }

  /**
   * Plays a gentle 2-tone 440 Hz -> 370 Hz chime once for technical events (Leads Off / Disconnect).
   */
  triggerTechnicalAlert() {
    if (this._muted || !this.unlocked) return;
    this._playTone(440, 0.13, 0.14, 0);
    const id = setTimeout(() => {
      if (!this._muted) {
        this._playTone(370, 0.18, 0.14, 0);
      }
    }, 150);
    this._scheduledTimeoutIds.push(id);
  }

  /**
   * Toggles a 120-second ICU Audio Pause countdown; invokes onTickCallback(remainingSeconds) every second.
   * @param {function(number): void} [onTickCallback]
   * @returns {boolean} Current muted state after toggle
   */
  toggleMute2Min(onTickCallback) {
    if (typeof onTickCallback === "function") {
      this._onMuteTick = onTickCallback;
    }

    if (this._muted) {
      // Unmute immediately
      this._muted = false;
      this._muteRemaining = 0;
      if (this._muteTimerId !== null) {
        clearInterval(this._muteTimerId);
        this._muteTimerId = null;
      }
      if (this._onMuteTick) {
        this._onMuteTick(0);
      }
      return false;
    }

    // Activate 120-second audio pause
    this._muted = true;
    this._muteRemaining = 120;
    while (this._scheduledTimeoutIds.length > 0) {
      clearTimeout(this._scheduledTimeoutIds.pop());
    }

    if (this._onMuteTick) {
      this._onMuteTick(this._muteRemaining);
    }

    this._muteTimerId = setInterval(() => {
      this._muteRemaining -= 1;
      if (this._muteRemaining <= 0) {
        this._muted = false;
        this._muteRemaining = 0;
        clearInterval(this._muteTimerId);
        this._muteTimerId = null;
        if (this._onMuteTick) {
          this._onMuteTick(0);
        }
      } else if (this._onMuteTick) {
        this._onMuteTick(this._muteRemaining);
      }
    }, 1000);

    return true;
  }

  /**
   * Returns true if ICU 2-minute Audio Pause is currently active.
   * @returns {boolean}
   */
  isMuted() {
    return this._muted;
  }

  /**
   * Immediately halts all active alarm timers and mute countdowns.
   */
  stopAll() {
    this.alarmLevel = "none";
    this._clearAlarmTimers();
    if (this._muteTimerId !== null) {
      clearInterval(this._muteTimerId);
      this._muteTimerId = null;
    }
    this._muted = false;
    this._muteRemaining = 0;
  }
}

from __future__ import annotations

import os
from typing import Any

import joblib
import numpy as np
from sklearn.ensemble import RandomForestClassifier


class HybridAIClassifier:
    """Lightweight Random Forest + Clinical Rules ECG Rhythm & ST Classifier."""

    CLASS_LABELS: dict[int, str] = {
        0: "NORMAL SINUS RHYTHM",
        1: "SINUS TACHYCARDIA",
        2: "SINUS BRADYCARDIA",
        3: "ARRHYTHMIA / IRREGULAR RHYTHM",
        4: "ST SEGMENT ELEVATION RISK",
    }

    FEATURE_ORDER: tuple[str, ...] = (
        "bpm",
        "rr_mean_ms",
        "rr_std_ms",
        "rmssd_ms",
        "rr_cv",
        "qrs_ms",
        "st_mv",
        "r_amp_cv",
    )

    def __init__(self, model_path: str | None = None) -> None:
        self.model: RandomForestClassifier = RandomForestClassifier(
            n_estimators=40,
            max_depth=8,
            min_samples_leaf=2,
            random_state=42,
            n_jobs=1,
        )
        self.is_external_model: bool = False
        self._train_synthetic_mitbih_baseline()
        if model_path and os.path.isfile(model_path):
            self.load_external_model(model_path)

    def _train_synthetic_mitbih_baseline(self) -> None:
        """
        Generates a deterministic, physiologically grounded synthetic dataset
        modeled on MIT-BIH arrhythmia feature distributions across 5 classes
        and fits a compact RandomForestClassifier in <150 ms.
        """
        rng = np.random.default_rng(42)
        n_per_class = 160
        X_blocks: list[np.ndarray] = []
        y_blocks: list[np.ndarray] = []

        # Class 0: NORMAL SINUS RHYTHM (60-100 BPM, low RR CV, normal ST)
        bpm_0 = rng.uniform(61.0, 99.0, n_per_class)
        rr_mean_0 = 60000.0 / bpm_0
        rr_cv_0 = rng.uniform(0.015, 0.085, n_per_class)
        rr_std_0 = rr_mean_0 * rr_cv_0
        rmssd_0 = rr_std_0 * rng.uniform(0.85, 1.25, n_per_class)
        qrs_0 = rng.normal(88.0, 8.0, n_per_class).clip(68.0, 112.0)
        st_0 = rng.normal(0.01, 0.04, n_per_class).clip(-0.11, 0.11)
        ramp_0 = rng.uniform(0.02, 0.10, n_per_class)
        X_blocks.append(
            np.column_stack([bpm_0, rr_mean_0, rr_std_0, rmssd_0, rr_cv_0, qrs_0, st_0, ramp_0])
        )
        y_blocks.append(np.zeros(n_per_class, dtype=int))

        # Class 1: SINUS TACHYCARDIA (102-175 BPM, regular RR, normal ST)
        bpm_1 = rng.uniform(102.0, 175.0, n_per_class)
        rr_mean_1 = 60000.0 / bpm_1
        rr_cv_1 = rng.uniform(0.012, 0.080, n_per_class)
        rr_std_1 = rr_mean_1 * rr_cv_1
        rmssd_1 = rr_std_1 * rng.uniform(0.80, 1.15, n_per_class)
        qrs_1 = rng.normal(84.0, 7.5, n_per_class).clip(64.0, 108.0)
        st_1 = rng.normal(-0.01, 0.045, n_per_class).clip(-0.12, 0.12)
        ramp_1 = rng.uniform(0.02, 0.11, n_per_class)
        X_blocks.append(
            np.column_stack([bpm_1, rr_mean_1, rr_std_1, rmssd_1, rr_cv_1, qrs_1, st_1, ramp_1])
        )
        y_blocks.append(np.full(n_per_class, 1, dtype=int))

        # Class 2: SINUS BRADYCARDIA (34-58 BPM, regular RR, normal ST)
        bpm_2 = rng.uniform(34.0, 58.5, n_per_class)
        rr_mean_2 = 60000.0 / bpm_2
        rr_cv_2 = rng.uniform(0.018, 0.095, n_per_class)
        rr_std_2 = rr_mean_2 * rr_cv_2
        rmssd_2 = rr_std_2 * rng.uniform(0.90, 1.30, n_per_class)
        qrs_2 = rng.normal(92.0, 8.5, n_per_class).clip(70.0, 116.0)
        st_2 = rng.normal(0.00, 0.04, n_per_class).clip(-0.11, 0.11)
        ramp_2 = rng.uniform(0.02, 0.10, n_per_class)
        X_blocks.append(
            np.column_stack([bpm_2, rr_mean_2, rr_std_2, rmssd_2, rr_cv_2, qrs_2, st_2, ramp_2])
        )
        y_blocks.append(np.full(n_per_class, 2, dtype=int))

        # Class 3: ARRHYTHMIA / IRREGULAR RHYTHM (high RR variability / ectopic timing)
        bpm_3 = rng.uniform(48.0, 145.0, n_per_class)
        rr_mean_3 = 60000.0 / bpm_3
        rr_cv_3 = rng.uniform(0.165, 0.42, n_per_class)
        rr_std_3 = rr_mean_3 * rr_cv_3
        rmssd_3 = rr_std_3 * rng.uniform(1.05, 1.65, n_per_class)
        qrs_3 = rng.normal(104.0, 16.0, n_per_class).clip(74.0, 155.0)
        st_3 = rng.normal(0.00, 0.06, n_per_class).clip(-0.13, 0.13)
        ramp_3 = rng.uniform(0.08, 0.28, n_per_class)
        X_blocks.append(
            np.column_stack([bpm_3, rr_mean_3, rr_std_3, rmssd_3, rr_cv_3, qrs_3, st_3, ramp_3])
        )
        y_blocks.append(np.full(n_per_class, 3, dtype=int))

        # Class 4: ST SEGMENT ELEVATION RISK (significant ST deviation >= 0.15 mV or <= -0.15 mV)
        bpm_4 = rng.uniform(58.0, 118.0, n_per_class)
        rr_mean_4 = 60000.0 / bpm_4
        rr_cv_4 = rng.uniform(0.02, 0.11, n_per_class)
        rr_std_4 = rr_mean_4 * rr_cv_4
        rmssd_4 = rr_std_4 * rng.uniform(0.85, 1.25, n_per_class)
        qrs_4 = rng.normal(94.0, 10.0, n_per_class).clip(72.0, 128.0)
        st_signs = rng.choice([-1.0, 1.0, 1.0, 1.0], size=n_per_class)
        st_4 = st_signs * rng.uniform(0.16, 0.45, n_per_class)
        ramp_4 = rng.uniform(0.03, 0.14, n_per_class)
        X_blocks.append(
            np.column_stack([bpm_4, rr_mean_4, rr_std_4, rmssd_4, rr_cv_4, qrs_4, st_4, ramp_4])
        )
        y_blocks.append(np.full(n_per_class, 4, dtype=int))

        X = np.vstack(X_blocks)
        y = np.concatenate(y_blocks)
        self.model.fit(X, y)

    def load_external_model(self, path: str) -> None:
        """Loads a pre-trained joblib model artifact without changing caller contracts."""
        loaded: Any = joblib.load(path)
        if isinstance(loaded, dict) and "model" in loaded:
            self.model = loaded["model"]
        else:
            self.model = loaded
        self.is_external_model = True

    def classify(self, features: dict, min_hr: int = 60, max_hr: int = 100) -> dict:
        """
        Combines RandomForest probability inference with deterministic clinical guardrails.
        """
        bpm = float(features.get("bpm", 75.0))
        rr_mean_ms = float(features.get("rr_mean_ms", 60000.0 / max(bpm, 25.0)))
        rr_std_ms = float(features.get("rr_std_ms", 25.0))
        rmssd_ms = float(features.get("rmssd_ms", 28.0))
        rr_cv = float(features.get("rr_cv", rr_std_ms / max(rr_mean_ms, 1.0)))
        qrs_ms = float(features.get("qrs_ms", 88.0))
        st_mv = float(features.get("st_mv", 0.0))
        r_amp_cv = float(features.get("r_amp_cv", 0.04))

        vec = np.array(
            [[bpm, rr_mean_ms, rr_std_ms, rmssd_ms, rr_cv, qrs_ms, st_mv, r_amp_cv]],
            dtype=np.float64,
        )

        probs = self.model.predict_proba(vec)[0]
        pred_idx = int(np.argmax(probs))
        ml_label = self.CLASS_LABELS.get(pred_idx, "NORMAL SINUS RHYTHM")
        ml_conf = float(probs[pred_idx]) * 100.0

        # Clinical condition flags
        is_brady = bpm < float(min_hr)
        is_tachy = bpm > float(max_hr)
        is_extreme_brady = bpm < 45.0
        is_extreme_tachy = bpm > 130.0
        is_arrhythmia = (rr_cv >= 0.16) or (rmssd_ms >= 135.0 and rr_cv >= 0.13)
        is_st_risk = abs(st_mv) >= 0.15

        # Hybrid arbitration: synchronize ML probabilities with hard clinical boundaries
        reasons: list[str] = []
        chosen_label = ml_label
        chosen_conf = ml_conf

        if is_arrhythmia:
            chosen_label = "ARRHYTHMIA / IRREGULAR RHYTHM"
            rule_boost = min(99.0, 78.0 + (rr_cv - 0.16) * 110.0)
            chosen_conf = max( float(probs[3]) * 100.0, rule_boost )
            reasons.append(
                f"Irregular R-R cadence detected (RR CV {rr_cv * 100:.1f}%, RMSSD {rmssd_ms:.0f} ms)"
            )
            if is_st_risk:
                reasons.append(f"concurrent ST deviation ({st_mv:+.2f} mV)")
        elif is_st_risk:
            chosen_label = "ST SEGMENT ELEVATION RISK"
            rule_boost = min(98.0, 80.0 + (abs(st_mv) - 0.15) * 90.0)
            chosen_conf = max(float(probs[4]) * 100.0, rule_boost)
            direction = "elevation" if st_mv > 0 else "depression"
            reasons.append(
                f"ST segment {direction} of {st_mv:+.2f} mV exceeds +/-0.15 mV screening threshold"
            )
            if is_tachy:
                reasons.append(f"HR elevated at {bpm:.0f} BPM")
            elif is_brady:
                reasons.append(f"HR low at {bpm:.0f} BPM")
        elif is_tachy:
            chosen_label = "SINUS TACHYCARDIA"
            margin = (bpm - float(max_hr)) / max(float(max_hr), 1.0)
            rule_boost = min(98.0, 82.0 + margin * 60.0)
            chosen_conf = max(float(probs[1]) * 100.0, rule_boost)
            if is_extreme_tachy:
                reasons.append(
                    f"Critical tachycardia ({bpm:.0f} BPM > 130 BPM high-alarm threshold)"
                )
            else:
                reasons.append(
                    f"Heart rate {bpm:.0f} BPM exceeds upper limit ({max_hr} BPM) with regular rhythm"
                )
        elif is_brady:
            chosen_label = "SINUS BRADYCARDIA"
            margin = (float(min_hr) - bpm) / max(float(min_hr), 1.0)
            rule_boost = min(98.0, 82.0 + margin * 65.0)
            chosen_conf = max(float(probs[2]) * 100.0, rule_boost)
            if is_extreme_brady:
                reasons.append(
                    f"Critical bradycardia ({bpm:.0f} BPM < 45 BPM high-alarm threshold)"
                )
            else:
                reasons.append(
                    f"Heart rate {bpm:.0f} BPM below lower limit ({min_hr} BPM) with regular rhythm"
                )
        else:
            chosen_label = "NORMAL SINUS RHYTHM"
            chosen_conf = max(float(probs[0]) * 100.0, 86.0)
            reasons.append(
                f"Regular sinus cadence ({bpm:.0f} BPM, RR CV {rr_cv * 100:.1f}%, ST {st_mv:+.2f} mV)"
            )

        # Alarm priority determination
        if is_arrhythmia or is_extreme_brady or is_extreme_tachy:
            alarm_level = "high"
        elif is_brady or is_tachy or is_st_risk:
            alarm_level = "medium"
        else:
            alarm_level = "none"

        confidence_pct = int(np.clip(round(chosen_conf), 50, 99))

        return {
            "label": chosen_label,
            "confidence_pct": confidence_pct,
            "alarm_level": alarm_level,
            "reason": "; ".join(reasons) + ".",
            "bradycardia": bool(is_brady),
            "tachycardia": bool(is_tachy),
            "arrhythmia": bool(is_arrhythmia),
        }

from __future__ import annotations

import threading
import time
from collections import deque

import numpy as np
from scipy import signal

from ai_classifier import HybridAIClassifier


class ECGProcessor:
    """
    Dual-Path Biomedical DSP Pipeline for 250 Hz AD8232 Single-Lead ECG.
    - Fast Path (10 Hz, 25 samples): Stateful causal bandpass + 50 Hz notch + real-time R-peak markers.
    - Slow Path (1 Hz, 6-second window): Zero-phase filtering, Multi-Factor SQI, Pan-Tompkins QRS,
      Q-S duration, J+60ms ST deviation, and Hybrid AI classification.
    """

    AD8232_MID_MV: float = 1650.0
    AD8232_GAIN_SCALE: float = 1100.0

    def __init__(self, fs: int = 250, classifier: HybridAIClassifier | None = None) -> None:
        self.fs: int = int(fs)
        self.classifier: HybridAIClassifier = classifier if classifier is not None else HybridAIClassifier()

        self.min_hr: int = 60
        self.max_hr: int = 100

        self._lock = threading.Lock()

        # Filter coefficients
        nyq = 0.5 * self.fs
        # 4th-order Butterworth bandpass 0.5 - 40.0 Hz
        self._bp_sos = signal.butter(
            4, [0.5 / nyq, 40.0 / nyq], btype="bandpass", output="sos"
        )
        self._bp_zi_template = signal.sosfilt_zi(self._bp_sos)

        # 50 Hz IIR Notch filter (Q = 30)
        self._notch_b, self._notch_a = signal.iirnotch(50.0, 30.0, fs=self.fs)
        self._notch_zi_template = signal.lfilter_zi(self._notch_b, self._notch_a)

        # Pan-Tompkins QRS bandpass (5 - 15 Hz, 2nd-order zero-phase)
        self._pt_sos = signal.butter(
            2, [5.0 / nyq, 15.0 / nyq], btype="bandpass", output="sos"
        )

        # High-frequency EMG noise estimation high-pass (> 40 Hz)
        self._hf_sos = signal.butter(
            3, 40.0 / nyq, btype="highpass", output="sos"
        )

        # Rolling buffers
        self._window_len: int = self.fs * 6          # 1500 samples (6 seconds)
        self._session_len: int = self.fs * 300       # 75000 samples (5 minutes)

        self._raw_buffer: deque[float] = deque(maxlen=self._window_len)
        self._clean_buffer: deque[float] = deque(maxlen=self._window_len)
        self._session_buffer: deque[dict] = deque(maxlen=self._session_len)

        # Filter states and runtime trackers
        self._bp_zi: np.ndarray | None = None
        self._notch_zi: np.ndarray | None = None
        self._leads_off: bool = False
        self._calibrating_samples_left: int = 0

        # Fast-path R-peak detector state
        self._sample_counter: int = 0
        self._last_fast_r_sample: int = -self.fs
        self._fast_peak_env: float = 0.75
        self._prev_prev_val: float = 0.0
        self._prev_val: float = 0.0

        # Slow-path EMA and status trackers
        self._ema_bpm: float | None = None
        self._ema_rr_ms: float | None = None
        self._latest_bpm_int: int | None = None
        self._latest_status: str = "CALIBRATING BASELINE..."

    def reset(self) -> None:
        """Clears buffers, filter states, EMA state, and calibration timer."""
        with self._lock:
            self._raw_buffer.clear()
            self._clean_buffer.clear()
            self._session_buffer.clear()
            self._bp_zi = None
            self._notch_zi = None
            self._leads_off = False
            self._calibrating_samples_left = 0
            self._sample_counter = 0
            self._last_fast_r_sample = -self.fs
            self._fast_peak_env = 0.75
            self._prev_prev_val = 0.0
            self._prev_val = 0.0
            self._ema_bpm = None
            self._ema_rr_ms = None
            self._latest_bpm_int = None
            self._latest_status = "CALIBRATING BASELINE..."

    def reset_filters(self) -> None:
        """Resets only causal filter states (used on packet-loss gaps > 3)."""
        with self._lock:
            self._bp_zi = None
            self._notch_zi = None
            self._prev_prev_val = 0.0
            self._prev_val = 0.0

    def set_hr_limits(self, min_hr: int, max_hr: int) -> None:
        """Updates patient-specific bradycardia/tachycardia alarm limits."""
        with self._lock:
            self.min_hr = max(30, min(int(min_hr), 110))
            self.max_hr = max(self.min_hr + 10, min(int(max_hr), 220))

    def ingest_batch(self, raw_mv: list[float], leads_off: bool = False) -> dict:
        """
        Fast Path (10 Hz, 25 samples per batch).
        Converts raw ADC mV to physiological mV, applies stateful causal filtering,
        manages post-leads-off 2.0-second calibration window, and detects instant R-peaks.
        """
        if not raw_mv:
            return {
                "samples": [],
                "r_peak_offsets": [],
                "leads_off": bool(leads_off),
                "calibrating": False,
            }

        with self._lock:
            was_leads_off = self._leads_off
            self._leads_off = bool(leads_off)

            # Transition True -> False triggers a 2.0-second (500-sample) post-leads-off calibration window
            if was_leads_off and not self._leads_off:
                self._calibrating_samples_left = int(2.0 * self.fs)
                self._bp_zi = None
                self._notch_zi = None

            if self._leads_off:
                self._bp_zi = None
                self._notch_zi = None
                zeros = [0.0 for _ in raw_mv]
                now_ts = time.time()
                dt = 1.0 / self.fs
                n_samples = len(raw_mv)
                for idx, r_val in enumerate(raw_mv):
                    self._raw_buffer.append(float(r_val))
                    self._clean_buffer.append(0.0)
                    sample_ts = now_ts - (n_samples - 1 - idx) * dt
                    self._session_buffer.append(
                        {
                            "ts": round(sample_ts, 4),
                            "ecg_mv": 0.0,
                            "bpm": None,
                            "status": "LEADS OFF",
                        }
                    )
                self._sample_counter += n_samples
                return {
                    "samples": zeros,
                    "r_peak_offsets": [],
                    "leads_off": True,
                    "calibrating": False,
                }

            raw_arr = np.asarray(raw_mv, dtype=np.float64)
            scaled_arr = (raw_arr - self.AD8232_MID_MV) / self.AD8232_GAIN_SCALE
            scaled_arr = np.clip(scaled_arr, -3.0, 3.5)

            # Initialize filter states cleanly on the first valid sample to avoid startup ringing
            if self._notch_zi is None or self._bp_zi is None:
                first_val = float(scaled_arr[0])
                self._notch_zi = self._notch_zi_template * first_val
                self._bp_zi = self._bp_zi_template * first_val

            notched_arr, self._notch_zi = signal.lfilter(
                self._notch_b, self._notch_a, scaled_arr, zi=self._notch_zi
            )
            cleaned_arr, self._bp_zi = signal.sosfilt(
                self._bp_sos, notched_arr, zi=self._bp_zi
            )
            cleaned_arr = np.clip(cleaned_arr, -2.0, 2.8)

            calibrating = self._calibrating_samples_left > 0
            if self._calibrating_samples_left > 0:
                self._calibrating_samples_left = max(
                    0, self._calibrating_samples_left - len(raw_mv)
                )
                calibrating = True

            # Real-time lightweight R-peak offset detection within the 25-sample batch
            r_peak_offsets: list[int] = []
            refractory_samples = int(0.22 * self.fs)  # 55 samples (220 ms)

            now_ts = time.time()
            dt = 1.0 / self.fs
            n_samples = len(cleaned_arr)
            cleaned_list: list[float] = []

            for i in range(n_samples):
                val = float(cleaned_arr[i])
                val_rounded = round(val, 4)
                cleaned_list.append(val_rounded)

                self._raw_buffer.append(float(raw_arr[i]))
                self._clean_buffer.append(val_rounded)

                sample_ts = now_ts - (n_samples - 1 - i) * dt
                self._session_buffer.append(
                    {
                        "ts": round(sample_ts, 4),
                        "ecg_mv": val_rounded,
                        "bpm": self._latest_bpm_int,
                        "status": self._latest_status,
                    }
                )

                # Adaptive causal peak detector (checks if prev_val was a local maximum)
                global_idx = self._sample_counter + i
                if not calibrating and len(self._clean_buffer) > int(0.5 * self.fs):
                    self._fast_peak_env = max(0.35, 0.992 * self._fast_peak_env)
                    if val > self._fast_peak_env:
                        self._fast_peak_env = 0.7 * self._fast_peak_env + 0.3 * val

                    thresh = max(0.32, 0.52 * self._fast_peak_env)
                    if (
                        self._prev_val > thresh
                        and self._prev_val >= self._prev_prev_val
                        and self._prev_val > val
                        and (global_idx - 1 - self._last_fast_r_sample) >= refractory_samples
                    ):
                        peak_offset = max(0, i - 1)
                        if not r_peak_offsets or (peak_offset - r_peak_offsets[-1]) >= refractory_samples:
                            r_peak_offsets.append(peak_offset)
                            self._last_fast_r_sample = global_idx - 1

                self._prev_prev_val = self._prev_val
                self._prev_val = val

            self._sample_counter += n_samples

            return {
                "samples": cleaned_list,
                "r_peak_offsets": r_peak_offsets,
                "leads_off": False,
                "calibrating": bool(calibrating),
            }

    def _compute_sqi(self, raw_arr: np.ndarray, notched_scaled: np.ndarray, zero_phase_ecg: np.ndarray) -> int:
        """
        Computes Multi-Factor Signal Quality Index (0 - 100%) from:
        1. ADC Rail Saturation (<= 50 mV or >= 3250 mV)
        2. Flatline variance check
        3. High-frequency EMG noise energy (> 40 Hz) vs QRS band energy (5 - 25 Hz)
        """
        if len(raw_arr) == 0:
            return 0

        # 1. Rail saturation check
        sat_mask = (raw_arr <= 50.0) | (raw_arr >= 3250.0)
        sat_ratio = float(np.mean(sat_mask))

        # 2. Flatline check
        ecg_std = float(np.std(zero_phase_ecg))
        ecg_ptp = float(np.ptp(zero_phase_ecg))
        if ecg_std < 0.025 or ecg_ptp < 0.15:
            return max(0, int(round(15.0 * (ecg_std / 0.025))))

        # 3. High-frequency EMG noise (>40 Hz on notch-filtered signal) vs QRS energy (5-25 Hz)
        qrs_sig = signal.sosfiltfilt(self._pt_sos, notched_scaled)
        hf_sig = signal.sosfiltfilt(self._hf_sos, notched_scaled)

        qrs_energy = float(np.mean(qrs_sig ** 2)) + 1e-9
        hf_energy = float(np.mean(hf_sig ** 2))
        hf_to_qrs_ratio = hf_energy / qrs_energy

        score = 100.0
        # Penalize ADC rail saturation heavily
        score -= min(65.0, sat_ratio * 380.0)
        # Penalize broadband high-frequency EMG noise
        if hf_to_qrs_ratio > 0.08:
            score -= min(75.0, (hf_to_qrs_ratio - 0.08) * 95.0)
        # Penalize excessive peak-to-peak erratic excursions (> 3.2 mV)
        if ecg_ptp > 3.2:
            score -= min(30.0, (ecg_ptp - 3.2) * 18.0)

        return int(np.clip(round(score), 0, 100))

    def _detect_r_peaks_pan_tompkins(self, zero_phase_ecg: np.ndarray) -> np.ndarray:
        """
        Pan-Tompkins R-Peak Detection:
        Bandpass (5-15 Hz) -> Derivative -> Squaring -> 150 ms Moving-Window Integration ->
        Adaptive thresholding with 220 ms refractory period -> Snapping to exact R-peak maxima.
        """
        n = len(zero_phase_ecg)
        if n < int(1.5 * self.fs):
            return np.array([], dtype=int)

        # 1. Bandpass 5-15 Hz
        qrs_band = signal.sosfiltfilt(self._pt_sos, zero_phase_ecg)

        # 2. Five-point derivative
        deriv = np.gradient(qrs_band)

        # 3. Squaring
        squared = deriv ** 2

        # 4. 150 ms moving-window integration
        win_len = max(3, int(0.150 * self.fs))
        kernel = np.ones(win_len, dtype=np.float64) / float(win_len)
        mwi = np.convolve(squared, kernel, mode="same")

        # 5. Adaptive threshold + 220 ms refractory period
        refractory = int(0.220 * self.fs)
        p75 = float(np.percentile(mwi, 75))
        p95 = float(np.percentile(mwi, 96))
        threshold = max(1e-6, p75 + 0.30 * (p95 - p75))

        candidate_peaks, _ = signal.find_peaks(
            mwi,
            height=threshold,
            distance=refractory,
        )

        if len(candidate_peaks) == 0:
            return np.array([], dtype=int)

        # 6. Snap each candidate to exact local maximum in zero-phase filtered ECG
        search_half = int(0.085 * self.fs)  # +/- 85 ms search window
        snapped_peaks: list[int] = []
        edge_margin = int(0.08 * self.fs)

        for pk in candidate_peaks:
            left = max(0, int(pk) - search_half)
            right = min(n, int(pk) + search_half + 1)
            if right <= left:
                continue
            local_idx = left + int(np.argmax(zero_phase_ecg[left:right]))
            if local_idx < edge_margin or local_idx >= (n - edge_margin):
                continue
            if zero_phase_ecg[local_idx] < 0.22:
                continue
            if snapped_peaks and (local_idx - snapped_peaks[-1]) < refractory:
                if zero_phase_ecg[local_idx] > zero_phase_ecg[snapped_peaks[-1]]:
                    snapped_peaks[-1] = local_idx
            else:
                snapped_peaks.append(local_idx)

        return np.array(snapped_peaks, dtype=int)

    def _extract_qrs_and_st(
        self, zero_phase_ecg: np.ndarray, r_peaks: np.ndarray
    ) -> tuple[int, float, str]:
        """
        Measures Q-onset to S-offset (QRS duration in ms) and ST deviation at (J-point + 60 ms)
        relative to the PR isoelectric baseline (60-100 ms before Q-onset).
        """
        n = len(zero_phase_ecg)
        qrs_durations_ms: list[float] = []
        st_values_mv: list[float] = []

        q_search = int(0.055 * self.fs)       # 55 ms before R for Q trough
        q_onset_max = int(0.060 * self.fs)    # max 60 ms before R for Q-onset
        q_onset_min = int(0.032 * self.fs)    # min 32 ms before R for Q-onset

        s_search = int(0.060 * self.fs)       # 60 ms after R for S trough
        s_offset_max = int(0.072 * self.fs)   # max 72 ms after R for J-point
        s_offset_min = int(0.036 * self.fs)   # min 36 ms after R for J-point

        st_offset_samples = int(0.060 * self.fs)  # J-point + 60 ms (15 samples)
        pr_start_offset = int(0.100 * self.fs)    # 100 ms before Q-onset
        pr_end_offset = int(0.060 * self.fs)      # 60 ms before Q-onset

        for r_idx_raw in r_peaks:
            r_idx = int(r_idx_raw)
            if r_idx - q_onset_max - pr_start_offset < 0:
                continue
            if r_idx + s_offset_max + st_offset_samples + 2 >= n:
                continue

            # Locate Q trough before R-peak
            q_left = max(0, r_idx - q_search)
            q_min_idx = q_left + int(np.argmin(zero_phase_ecg[q_left : r_idx + 1]))

            #Estimate preliminary baseline ahead of Q wave
            pre_q_left = max(0, r_idx - int(0.120 * self.fs))
            pre_q_right = max(pre_q_left + 1, r_idx - int(0.065 * self.fs))
            iso_ref = float(np.median(zero_phase_ecg[pre_q_left:pre_q_right]))

            # Walk backward from Q trough to find Q-onset
            q_onset_idx = q_min_idx
            q_bound = max(0, r_idx - q_onset_max)
            q_depth = abs(float(zero_phase_ecg[q_min_idx]) - iso_ref)
            q_thresh = iso_ref - 0.15 * max(q_depth, 0.04)
            for idx in range(q_min_idx, q_bound - 1, -1):
                q_onset_idx = idx
                if zero_phase_ecg[idx] >= q_thresh:
                    break
            q_onset_idx = int(
                np.clip(q_onset_idx, r_idx - q_onset_max, r_idx - q_onset_min)
            )

            # Locate S trough after R-peak
            s_right = min(n, r_idx + s_search + 1)
            s_min_idx = r_idx + int(np.argmin(zero_phase_ecg[r_idx:s_right]))

            # Walk forward from S trough to find S-offset (J-point) where slope flattens or returns to baseline
            s_offset_idx = s_min_idx
            s_bound = min(n - 2, r_idx + s_offset_max)
            s_depth = abs(float(zero_phase_ecg[s_min_idx]) - iso_ref)
            s_thresh = iso_ref - 0.15 * max(s_depth, 0.04)
            for idx in range(s_min_idx + 1, s_bound + 1):
                s_offset_idx = idx
                slope = abs(float(zero_phase_ecg[idx + 1] - zero_phase_ecg[idx]))
                if zero_phase_ecg[idx] >= s_thresh or slope < 0.012:
                    break
            s_offset_idx = int(
                np.clip(s_offset_idx, r_idx + s_offset_min, r_idx + s_offset_max)
            )

            qrs_ms_beat = ((s_offset_idx - q_onset_idx) / float(self.fs)) * 1000.0
            qrs_durations_ms.append(float(np.clip(qrs_ms_beat, 64.0, 160.0)))

            # PR isoelectric baseline: 60-100 ms before Q-onset
            pr_l = max(0, q_onset_idx - pr_start_offset)
            pr_r = max(pr_l + 2, q_onset_idx - pr_end_offset)
            pr_baseline_mv = float(np.mean(zero_phase_ecg[pr_l:pr_r]))

            # ST measurement at J-point + 60 ms
            st_center = s_offset_idx + st_offset_samples
            st_l = max(0, st_center - 1)
            st_r = min(n, st_center + 2)
            st_point_mv = float(np.mean(zero_phase_ecg[st_l:st_r]))

            st_values_mv.append(st_point_mv - pr_baseline_mv)

        qrs_ms = int(round(float(np.median(qrs_durations_ms)))) if qrs_durations_ms else 88
        st_mv = round(float(np.median(st_values_mv)), 3) if st_values_mv else 0.0

        if st_mv >= 0.15:
            st_status = "ELEVATED (Screening)"
        elif st_mv <= -0.15:
            st_status = "DEPRESSED (Screening)"
        else:
            st_status = "NORMAL"

        return qrs_ms, st_mv, st_status

    def analyze_window(self) -> dict:
        """
        Slow Path (1 Hz).
        Copies the 6-second raw buffer safely, applies zero-phase sosfiltfilt,
        evaluates Multi-Factor SQI gatekeeper, runs Pan-Tompkins R-peak detection,
        computes all 11 clinical parameters, and invokes HybridAIClassifier.
        """
        with self._lock:
            leads_off = self._leads_off
            calibrating = self._calibrating_samples_left > 0
            raw_snapshot = np.asarray(self._raw_buffer, dtype=np.float64)
            min_hr = self.min_hr
            max_hr = self.max_hr

        # Gatekeeper 1: Leads Off
        if leads_off:
            with self._lock:
                self._latest_bpm_int = None
                self._latest_status = "LEADS OFF"
            return {
                "valid_signal": False,
                "status": "LEADS OFF",
                "alarm_level": "technical",
                "sqi": 0,
                "bpm": None,
                "rr_ms": None,
                "hrv_ms": None,
                "qrs_ms": None,
                "st_mv": None,
                "st_status": None,
                "ai_label": "LEADS OFF",
                "ai_confidence": None,
                "ai_reason": "Electrode contact lost (LO+ / LO- triggered).",
                "bradycardia": False,
                "tachycardia": False,
                "arrhythmia": False,
            }

        # Gatekeeper 2: Calibrating or < 2.5 seconds of signal
        min_samples_required = int(2.5 * self.fs)
        if calibrating or len(raw_snapshot) < min_samples_required:
            with self._lock:
                self._latest_bpm_int = None
                self._latest_status = "CALIBRATING BASELINE..."
            return {
                "valid_signal": False,
                "status": "CALIBRATING BASELINE...",
                "alarm_level": "none",
                "sqi": None,
                "bpm": None,
                "rr_ms": None,
                "hrv_ms": None,
                "qrs_ms": None,
                "st_mv": None,
                "st_status": None,
                "ai_label": "CALIBRATING BASELINE...",
                "ai_confidence": None,
                "ai_reason": "Acquiring post-contact stabilized baseline window.",
                "bradycardia": False,
                "tachycardia": False,
                "arrhythmia": False,
            }

        # Zero-Phase Filtering (sosfiltfilt bandpass 0.5-40 Hz + filtfilt 50 Hz notch)
        scaled_raw = (raw_snapshot - self.AD8232_MID_MV) / self.AD8232_GAIN_SCALE
        notched_zero = signal.filtfilt(self._notch_b, self._notch_a, scaled_raw)
        zero_phase_ecg = signal.sosfiltfilt(self._bp_sos, notched_zero)

        # Multi-Factor SQI
        sqi = self._compute_sqi(raw_snapshot, notched_zero, zero_phase_ecg)

        # Gatekeeper 3: Low SQI (< 60%)
        if sqi < 60:
            with self._lock:
                self._latest_bpm_int = None
                self._latest_status = "LOW SIGNAL QUALITY â€” HOLD STILL"
            return {
                "valid_signal": False,
                "status": "LOW SIGNAL QUALITY â€” HOLD STILL",
                "alarm_level": "technical",
                "sqi": int(sqi),
                "bpm": None,
                "rr_ms": None,
                "hrv_ms": None,
                "qrs_ms": None,
                "st_mv": None,
                "st_status": None,
                "ai_label": "LOW SIGNAL QUALITY â€” HOLD STILL",
                "ai_confidence": None,
                "ai_reason": f"Signal quality ({sqi}%) below 60% diagnostic threshold due to motion or EMG artifact.",
                "bradycardia": False,
                "tachycardia": False,
                "arrhythmia": False,
            }

        # Pan-Tompkins R-Peak Detection
        r_peaks = self._detect_r_peaks_pan_tompkins(zero_phase_ecg)
        if len(r_peaks) < 2:
            with self._lock:
                self._latest_bpm_int = None
                self._latest_status = "LOW SIGNAL QUALITY â€” HOLD STILL"
            return {
                "valid_signal": False,
                "status": "LOW SIGNAL QUALITY â€” HOLD STILL",
                "alarm_level": "technical",
                "sqi": int(min(sqi, 55)),
                "bpm": None,
                "rr_ms": None,
                "hrv_ms": None,
                "qrs_ms": None,
                "st_mv": None,
                "st_status": None,
                "ai_label": "LOW SIGNAL QUALITY â€” HOLD STILL",
                "ai_confidence": None,
                "ai_reason": "Insufficient valid QRS complexes detected in analysis window.",
                "bradycardia": False,
                "tachycardia": False,
                "arrhythmia": False,
            }

        rr_intervals_ms = (np.diff(r_peaks) / float(self.fs)) * 1000.0
        valid_rr = rr_intervals_ms[(rr_intervals_ms >= 260.0) & (rr_intervals_ms <= 2200.0)]

        if len(valid_rr) == 0:
            with self._lock:
                self._latest_bpm_int = None
                self._latest_status = "LOW SIGNAL QUALITY â€” HOLD STILL"
            return {
                "valid_signal": False,
                "status": "LOW SIGNAL QUALITY â€” HOLD STILL",
                "alarm_level": "technical",
                "sqi": int(min(sqi, 55)),
                "bpm": None,
                "rr_ms": None,
                "hrv_ms": None,
                "qrs_ms": None,
                "st_mv": None,
                "st_status": None,
                "ai_label": "LOW SIGNAL QUALITY â€” HOLD STILL",
                "ai_confidence": None,
                "ai_reason": "Detected R-R intervals outside physiological bounds (260-2200 ms).",
                "bradycardia": False,
                "tachycardia": False,
                "arrhythmia": False,
            }

        median_rr_ms = float(np.median(valid_rr))
        raw_bpm = 60000.0 / median_rr_ms

        # EMA alpha=0.35 ONLY on UI readout metrics (BPM and RR)
        alpha = 0.35
        with self._lock:
            if self._ema_bpm is None or abs(raw_bpm - self._ema_bpm) > 28.0:
                self._ema_bpm = raw_bpm
                self._ema_rr_ms = median_rr_ms
            else:
                self._ema_bpm = alpha * raw_bpm + (1.0 - alpha) * self._ema_bpm
                self._ema_rr_ms = alpha * median_rr_ms + (1.0 - alpha) * self._ema_rr_ms
            smoothed_bpm = int(round(self._ema_bpm))
            smoothed_rr_ms = int(round(self._ema_rr_ms))

        # HRV (RMSSD in ms) and RR statistics
        rr_mean_ms = float(np.mean(valid_rr))
        rr_std_ms = float(np.std(valid_rr)) if len(valid_rr) > 1 else 12.0
        if len(valid_rr) >= 2:
            succ_diffs = np.diff(valid_rr)
            rmssd_ms = float(np.sqrt(np.mean(succ_diffs ** 2)))
        else:
            rmssd_ms = rr_std_ms
        rr_cv = rr_std_ms / max(rr_mean_ms, 1.0)
        hrv_ms_int = int(round(np.clip(rmssd_ms, 1.0, 350.0)))

        # QRS duration and ST segment deviation
        qrs_ms, st_mv, st_status = self._extract_qrs_and_st(zero_phase_ecg, r_peaks)

        # R-peak amplitude stability
        r_amps = zero_phase_ecg[r_peaks]
        r_amp_mean = float(np.mean(np.abs(r_amps)))
        r_amp_cv = float(np.std(r_amps) / max(r_amp_mean, 0.1))

        features = {
            "bpm": float(smoothed_bpm),
            "rr_mean_ms": rr_mean_ms,
            "rr_std_ms": rr_std_ms,
            "rmssd_ms": rmssd_ms,
            "rr_cv": rr_cv,
            "qrs_ms": float(qrs_ms),
            "st_mv": float(st_mv),
            "r_amp_cv": r_amp_cv,
        }

        ai_res = self.classifier.classify(features, min_hr=min_hr, max_hr=max_hr)

        with self._lock:
            self._latest_bpm_int = smoothed_bpm
            self._latest_status = ai_res["label"]

        return {
            "valid_signal": True,
            "status": ai_res["label"],
            "alarm_level": ai_res["alarm_level"],
            "sqi": int(sqi),
            "bpm": smoothed_bpm,
            "rr_ms": smoothed_rr_ms,
            "hrv_ms": hrv_ms_int,
            "qrs_ms": int(qrs_ms),
            "st_mv": round(float(st_mv), 2),
            "st_status": st_status,
            "ai_label": ai_res["label"],
            "ai_confidence": int(ai_res["confidence_pct"]),
            "ai_reason": ai_res["reason"],
            "bradycardia": bool(ai_res["bradycardia"]),
            "tachycardia": bool(ai_res["tachycardia"]),
            "arrhythmia": bool(ai_res["arrhythmia"]),
        }

    def get_recent_snapshot(self, seconds: float = 5.0) -> list[float]:
        """Returns the most recent N seconds of cleaned ECG samples in mV."""
        n_samples = max(1, int(round(seconds * self.fs)))
        with self._lock:
            buf_list = list(self._clean_buffer)
        return buf_list[-n_samples:]

    def get_session_records(self) -> list[dict]:
        """
        Returns list of {"ts": float, "ecg_mv": float, "bpm": int|None, "status": str}
        from the 5-minute session ring buffer.
        """
        with self._lock:
            return list(self._session_buffer)

from __future__ import annotations

import math
import numpy as np


class ECGSimulator:
    """
    250 Hz Phase-Continuous Synthetic AD8232 ECG Waveform Generator.
    Synthesizes realistic P-QRS-T morphology centered at 1650.0 mV (AD8232 gain ~1100)
    with injected 50 Hz AC mains hum, 0.25 Hz respiratory baseline wander, and Gaussian noise.
    """

    AD8232_MID_MV: float = 1650.0
    AD8232_GAIN_SCALE: float = 1100.0

    SUPPORTED_MODES: tuple[str, ...] = (
        "normal",
        "tachycardia",
        "bradycardia",
        "arrhythmia",
        "st_elevation",
        "motion_artifact",
        "leads_off",
    )

    def __init__(self, fs: int = 250) -> None:
        self.fs: int = int(fs)
        self._rng = np.random.default_rng(2026)
        self._t_sec: float = 0.0
        self._beat_elapsed_sec: float = 0.0
        self._current_rr_sec: float = 60.0 / 74.0
        self._arrhythmia_pattern_idx: int = 0
        self._arrhythmia_rr_factors: tuple[float, ...] = (
            1.00,
            0.62,
            1.36,
            0.94,
            0.58,
            1.42,
            0.78,
            1.28,
        )

    def reset(self) -> None:
        """Resets simulator phase, clocks, and deterministic random stream."""
        self._rng = np.random.default_rng(2026)
        self._t_sec = 0.0
        self._beat_elapsed_sec = 0.0
        self._current_rr_sec = 60.0 / 74.0
        self._arrhythmia_pattern_idx = 0

    def _next_rr_interval_sec(self, mode: str) -> float:
        """Selects the next beat-to-beat RR interval in seconds for the active mode."""
        if mode == "tachycardia":
            base_bpm = 126.0 + float(self._rng.normal(0.0, 1.2))
            return 60.0 / np.clip(base_bpm, 120.0, 132.0)
        if mode == "bradycardia":
            base_bpm = 48.0 + float(self._rng.normal(0.0, 0.8))
            return 60.0 / np.clip(base_bpm, 45.5, 51.0)
        if mode == "arrhythmia":
            base_rr = 60.0 / 78.0
            factor = self._arrhythmia_rr_factors[
                self._arrhythmia_pattern_idx % len(self._arrhythmia_rr_factors)
            ]
            self._arrhythmia_pattern_idx += 1
            jitter = float(self._rng.normal(0.0, 0.025))
            return float(np.clip(base_rr * factor + jitter, 0.38, 1.35))
        if mode == "st_elevation":
            base_bpm = 78.0 + float(self._rng.normal(0.0, 1.0))
            return 60.0 / np.clip(base_bpm, 74.0, 82.0)
        # Default ("normal" or "motion_artifact")
        base_bpm = 74.0 + float(self._rng.normal(0.0, 1.1))
        return 60.0 / np.clip(base_bpm, 70.0, 78.0)

    @staticmethod
    def _gaussian_wave(dt: float, center: float, width: float, amp: float) -> float:
        z = (dt - center) / width
        if abs(z) > 6.0:
            return 0.0
        return amp * math.exp(-0.5 * z * z)

    def _evaluate_pqrst_mv(self, dt_from_r: float, mode: str) -> float:
        """
        Evaluates clean physiological ECG voltage (in mV) at time offset `dt_from_r` (seconds)
        relative to the current beat's R-peak center.
        """
        # P wave (-160 ms before R)
        p_wave = self._gaussian_wave(dt_from_r, -0.155, 0.022, 0.16)
        # Q wave (-22 ms before R)
        q_wave = self._gaussian_wave(dt_from_r, -0.021, 0.0075, -0.15)
        # R peak (0 ms) -> +1.28 mV peak so filtered output lands in +1.0 to +1.4 mV
        r_wave = self._gaussian_wave(dt_from_r, 0.000, 0.0095, 1.28)
        # S wave (+22 ms after R)
        s_wave = self._gaussian_wave(dt_from_r, 0.022, 0.0085, -0.26)
        # T wave (+200 ms after R)
        t_wave = self._gaussian_wave(dt_from_r, 0.195, 0.036, 0.28)

        val = p_wave + q_wave + r_wave + s_wave + t_wave

        # Elevated ST plateau in "st_elevation" mode: J-point (+38 ms) through T-wave onset (+175 ms)
        if mode == "st_elevation":
            st_plateau = 0.34 / (
                (1.0 + math.exp(-(dt_from_r - 0.034) / 0.006))
                * (1.0 + math.exp((dt_from_r - 0.175) / 0.014))
            )
            val += st_plateau

        return val

    def generate_batch(self, mode: str, n: int = 25) -> tuple[list[float], bool]:
        """
        Generates `n` raw AD8232 millivolt samples (0 - 3300 mV, centered at 1650.0 mV)
        and the `leads_off` boolean flag.
        """
        normalized_mode = (mode or "normal").strip().lower()
        if normalized_mode not in self.SUPPORTED_MODES:
            normalized_mode = "normal"

        n_samples = max(1, int(n))
        dt_step = 1.0 / float(self.fs)

        if normalized_mode == "leads_off":
            samples: list[float] = []
            for _ in range(n_samples):
                self._t_sec += dt_step
                drift = 45.0 * math.sin(2.0 * math.pi * 1.5 * self._t_sec)
                noise = float(self._rng.normal(0.0, 18.0))
                samples.append(round(float(np.clip(self.AD8232_MID_MV + drift + noise, 0.0, 3300.0)), 2))
            return samples, True

        batch: list[float] = []
        for _ in range(n_samples):
            self._t_sec += dt_step
            self._beat_elapsed_sec += dt_step

            if self._beat_elapsed_sec >= self._current_rr_sec:
                self._beat_elapsed_sec -= self._current_rr_sec
                self._current_rr_sec = self._next_rr_interval_sec(normalized_mode)

            # Place R-peak at 38% of the current RR cycle so P-QRS-T stays intact
            r_center_sec = max(0.20, min(0.32, 0.38 * self._current_rr_sec))
            dt_from_r = self._beat_elapsed_sec - r_center_sec

            clean_phys_mv = self._evaluate_pqrst_mv(dt_from_r, normalized_mode)

            # Inject 0.25 Hz respiratory baseline wander + 50 Hz AC mains hum + light Gaussian noise
            wander_mv = 0.08 * math.sin(2.0 * math.pi * 0.25 * self._t_sec)
            ac_hum_mv = 0.06 * math.sin(2.0 * math.pi * 50.0 * self._t_sec)
            gaussian_noise_mv = float(self._rng.normal(0.0, 0.016))

            total_phys_mv = clean_phys_mv + wander_mv + ac_hum_mv + gaussian_noise_mv

            if normalized_mode == "motion_artifact":
                # Heavy broadband high-frequency EMG (45-110 Hz) + large movement swings
                emg_hf_mv = (
                    0.75 * math.sin(2.0 * math.pi * 63.0 * self._t_sec)
                    + 0.65 * math.cos(2.0 * math.pi * 87.0 * self._t_sec)
                    + float(self._rng.normal(0.0, 0.85))
                )
                motion_swing_mv = 0.95 * math.sin(2.0 * math.pi * 3.2 * self._t_sec)
                total_phys_mv += emg_hf_mv + motion_swing_mv

            raw_adc_mv = self.AD8232_MID_MV + total_phys_mv * self.AD8232_GAIN_SCALE

            # Inject periodic ADC rail spikes in motion_artifact mode so saturation penalty triggers
            if normalized_mode == "motion_artifact":
                spike_roll = float(self._rng.random())
                if spike_roll < 0.14:
                    raw_adc_mv = 15.0
                elif spike_roll < 0.28:
                    raw_adc_mv = 3285.0

            raw_adc_mv = float(np.clip(raw_adc_mv, 0.0, 3300.0))
            batch.append(round(raw_adc_mv, 2))

        return batch, False

class HeartixHub:
    """
    Central Real-Time Telemetry Hub managing:
    - Single active ESP32 hardware stream (with automatic reconnect takeover & priority)
    - Multiple concurrent browser WebSocket viewers
    - Dual-path ECGProcessor & 250 Hz ECGSimulator state
    - Sequence gap tracking, packet-loss telemetry, and 25-event clinical abnormality memory
    """

    def __init__(self) -> None:
        self.classifier = HybridAIClassifier()
        self.processor = ECGProcessor(fs=250, classifier=self.classifier)
        self.simulator = ECGSimulator(fs=250)

        self.esp32_ws: WebSocket | None = None
        self.esp32_connected: bool = False
        self.last_esp32_packet_ts: float = 0.0
        self.disconnected_mid_session: bool = False

        # Simulator MUST default to None (OFF) on server boot
        self.sim_mode: str | None = None

        self.web_clients: set[WebSocket] = set()

        # Rolling sequence & packet telemetry over a 10-second window
        self.last_seq: int | None = None
        self._packet_window: deque[tuple[float, int, int, int]] = deque(maxlen=250)
        self.packet_loss_pct: float = 0.0
        self.effective_hz: float = 0.0

        # Clinical abnormality events (capped at 25 in-memory records)
        self.events: deque[dict[str, Any]] = deque(maxlen=25)
        self.last_event_ts_by_label: dict[str, float] = {}

        # Active patient metadata
        self.patient: dict[str, Any] = {
            "name": "",
            "age": None,
            "gender": "",
            "patient_id": "",
            "bed": "",
            "min_hr": 60,
            "max_hr": 100,
            "notes": "",
        }

    def reset_telemetry(self) -> None:
        """Clears sequence counters and rolling packet-loss / sample-rate statistics."""
        self.last_seq = None
        self._packet_window.clear()
        self.packet_loss_pct = 0.0
        self.effective_hz = 0.0

    def record_esp32_packet(self, seq: int | None, sample_count: int) -> None:
        """
        Tracks uint16 sequence wraparound (0-65535), computes rolling packet_loss_pct
        and effective_hz, and resets causal DSP filters if a packet gap > 3 occurs.
        """
        now = time.monotonic()
        lost = 0

        if seq is not None:
            seq_u16 = int(seq) & 0xFFFF
            if self.last_seq is not None:
                delta = (seq_u16 - self.last_seq) & 0xFFFF
                if delta == 0:
                    lost = 0
                elif delta == 1:
                    lost = 0
                elif 1 < delta <= 1000:
                    lost = delta - 1
                    if lost > 3:
                        self.processor.reset_filters()
                else:
                    # Large jump indicates ESP32 reboot or counter reset
                    lost = 0
                    self.processor.reset_filters()
            self.last_seq = seq_u16

        self._packet_window.append((now, 1, lost, sample_count))

        # Prune telemetry older than 10.0 seconds
        cutoff = now - 10.0
        while self._packet_window and self._packet_window[0][0] < cutoff:
            self._packet_window.popleft()

        total_recv = sum(item[1] for item in self._packet_window)
        total_lost = sum(item[2] for item in self._packet_window)
        total_expected = total_recv + total_lost

        if total_expected > 0:
            self.packet_loss_pct = round((total_lost / float(total_expected)) * 100.0, 2)
        else:
            self.packet_loss_pct = 0.0

        if len(self._packet_window) >= 2:
            elapsed = now - self._packet_window[0][0]
            if elapsed >= 0.15:
                samples_after_first = sum(item[3] for item in list(self._packet_window)[1:])
                self.effective_hz = round(min(260.0, samples_after_first / elapsed), 1)
            else:
                self.effective_hz = 250.0
        elif len(self._packet_window) == 1:
            self.effective_hz = 250.0
        else:
            self.effective_hz = 0.0

    def build_state_payload(self) -> dict[str, Any]:
        """Constructs the canonical state synchronization message for web clients."""
        return {
            "type": "state",
            "esp32_connected": self.esp32_connected,
            "sim_mode": self.sim_mode,
            "packet_loss_pct": self.packet_loss_pct,
            "effective_hz": self.effective_hz,
            "disconnected_mid_session": self.disconnected_mid_session,
            "patient": self.patient,
        }

    def build_disconnect_metrics_payload(self) -> dict[str, Any]:
        """Constructs offline telemetry metrics with a low-priority technical disconnect alarm."""
        return {
            "type": "metrics",
            "valid_signal": False,
            "status": "ESP32 DISCONNECTED â€” CHECK DEVICE",
            "alarm_level": "low",
            "sqi": 0,
            "bpm": None,
            "rr_ms": None,
            "hrv_ms": None,
            "qrs_ms": None,
            "st_mv": None,
            "st_status": None,
            "ai_label": "ESP32 DISCONNECTED",
            "ai_confidence": None,
            "ai_reason": "ESP32 hardware telemetry stream interrupted or timed out.",
            "bradycardia": False,
            "tachycardia": False,
            "arrhythmia": False,
            "event_count": len(self.events),
            "packet_loss_pct": self.packet_loss_pct,
            "effective_hz": self.effective_hz,
        }

    async def broadcast_json(self, payload: dict[str, Any]) -> None:
        """Broadcasts a JSON dictionary to all connected browser clients and prunes dead sockets."""
        if not self.web_clients:
            return
        dead_clients: list[WebSocket] = []
        message_text = json.dumps(payload)
        for ws in list(self.web_clients):
            try:
                await ws.send_text(message_text)
            except Exception:
                dead_clients.append(ws)
        for ws in dead_clients:
            self.web_clients.discard(ws)

    async def broadcast_state(self) -> None:
        """Broadcasts the current system state to all browser clients."""
        await self.broadcast_json(self.build_state_payload())

async def esp32_watchdog_loop() -> None:
    """
    Runs every 0.5s. If esp32_connected is True and no packet has arrived for 2.5 seconds,
    forcefully closes the dead ESP32 socket, resets processor/telemetry, and broadcasts
    updated state + offline metrics (with alarm_level="low").
    """
    while True:
        await asyncio.sleep(0.5)
        if hub.esp32_connected:
            elapsed = time.monotonic() - hub.last_esp32_packet_ts
            if elapsed > 2.5:
                dead_ws = hub.esp32_ws
                hub.esp32_ws = None
                hub.esp32_connected = False
                hub.disconnected_mid_session = True
                hub.processor.reset()
                hub.reset_telemetry()

                if dead_ws is not None:
                    try:
                        await dead_ws.close(code=1001, reason="Watchdog timeout (>2.5s silence)")
                    except Exception:
                        pass

                await hub.broadcast_state()
                await hub.broadcast_json(hub.build_disconnect_metrics_payload())


async def analysis_loop() -> None:
    """
    Runs every 1.0s. When esp32_connected OR sim_mode is active, executes
    processor.analyze_window() in a worker thread, logs medium/high alarm events
    with a 10-second per-label cooldown (capped at 25 events), and broadcasts metrics.
    """
    while True:
        await asyncio.sleep(1.0)
        if not (hub.esp32_connected or hub.sim_mode is not None):
            continue

        result = await asyncio.to_thread(hub.processor.analyze_window)

        if result.get("valid_signal") is True and result.get("alarm_level") in ("medium", "high"):
            ai_label = str(result.get("ai_label") or "ABNORMALITY")
            now_mono = time.monotonic()
            last_logged = hub.last_event_ts_by_label.get(ai_label, -999.0)
            if (now_mono - last_logged) >= 10.0:
                hub.last_event_ts_by_label[ai_label] = now_mono
                waveform_5s = await asyncio.to_thread(hub.processor.get_recent_snapshot, 5.0)
                event_record = {
                    "id": uuid.uuid4().hex[:10],
                    "timestamp": datetime.now(timezone.utc).isoformat(),
                    "source": "esp32" if hub.esp32_connected else f"simulator:{hub.sim_mode}",
                    "ai_label": ai_label,
                    "ai_confidence": result.get("ai_confidence"),
                    "ai_reason": result.get("ai_reason"),
                    "alarm_level": result.get("alarm_level"),
                    "bpm": result.get("bpm"),
                    "rr_ms": result.get("rr_ms"),
                    "hrv_ms": result.get("hrv_ms"),
                    "qrs_ms": result.get("qrs_ms"),
                    "st_mv": result.get("st_mv"),
                    "st_status": result.get("st_status"),
                    "sqi": result.get("sqi"),
                    "bradycardia": result.get("bradycardia"),
                    "tachycardia": result.get("tachycardia"),
                    "arrhythmia": result.get("arrhythmia"),
                    "patient_id": hub.patient.get("patient_id", ""),
                    "patient_name": hub.patient.get("name", ""),
                    "waveform_5s": waveform_5s,
                }
                hub.events.appendleft(event_record)

        metrics_payload = {
            "type": "metrics",
            **result,
            "event_count": len(hub.events),
            "packet_loss_pct": hub.packet_loss_pct,
            "effective_hz": hub.effective_hz,
        }
        await hub.broadcast_json(metrics_payload)


async def simulator_loop() -> None:
    """
    Runs every 0.1s (10 Hz). Active ONLY when not esp32_connected, sim_mode is set,
    and at least 1 web client is connected. Generates 25 samples at 250 Hz, feeds
    them through processor.ingest_batch(), and broadcasts ecg_batch.
    """
    next_tick = time.monotonic()
    while True:
        next_tick += 0.1
        sleep_dur = max(0.005, next_tick - time.monotonic())
        await asyncio.sleep(sleep_dur)

        if time.monotonic() - next_tick > 0.5:
            next_tick = time.monotonic()

        if hub.esp32_connected or hub.sim_mode is None or len(hub.web_clients) == 0:
            continue

        raw_mv, leads_off = hub.simulator.generate_batch(hub.sim_mode, 25)
        fast_out = hub.processor.ingest_batch(raw_mv, leads_off=leads_off)
        hub.packet_loss_pct = 0.0
        hub.effective_hz = 250.0

        await hub.broadcast_json(
            {
                "type": "ecg_batch",
                "source": "simulator",
                "mode": hub.sim_mode,
                "samples": fast_out["samples"],
                "r_peak_offsets": fast_out["r_peak_offsets"],
                "leads_off": fast_out["leads_off"],
                "calibrating": fast_out["calibrating"],
            }
        )


async def web_keepalive_loop() -> None:
    """
    Runs every 20s. Sends {"type": "ping"} to all connected /ws/web clients
    and prunes unresponsive connections.
    """
    while True:
        await asyncio.sleep(20.0)
        await hub.broadcast_json({"type": "ping"})


@asynccontextmanager
async def lifespan(app: FastAPI):
    tasks = [
        asyncio.create_task(esp32_watchdog_loop(), name="esp32_watchdog"),
        asyncio.create_task(analysis_loop(), name="analysis_loop"),
        asyncio.create_task(simulator_loop(), name="simulator_loop"),
        asyncio.create_task(web_keepalive_loop(), name="web_keepalive"),
    ]
    try:
        yield
    finally:
        for task in tasks:
            task.cancel()
        await asyncio.gather(*tasks, return_exceptions=True)

@app.post("/api/patient")
async def update_patient(patient: PatientModel) -> dict[str, Any]:
    min_hr = min(patient.min_hr, patient.max_hr - 10)
    max_hr = max(patient.max_hr, min_hr + 10)

    hub.patient = {
        "name": patient.name.strip(),
        "age": patient.age,
        "gender": patient.gender.strip(),
        "patient_id": patient.patient_id.strip(),
        "bed": patient.bed.strip(),
        "min_hr": min_hr,
        "max_hr": max_hr,
        "notes": patient.notes.strip(),
    }
    hub.processor.set_hr_limits(min_hr, max_hr)
    await hub.broadcast_state()
    return {"status": "ok", "patient": hub.patient}


@app.get("/api/export/csv")
async def export_csv() -> StreamingResponse:
    records = await asyncio.to_thread(hub.processor.get_session_records)
    patient = hub.patient
    now_utc = datetime.now(timezone.utc)
    ts_str = now_utc.strftime("%Y%m%d_%H%M%S")

    raw_pid = str(patient.get("patient_id") or "anonymous").strip()
    safe_pid = re.sub(r"[^A-Za-z0-9_-]+", "_", raw_pid).strip("_") or "anonymous"
    filename = f"heartix_{safe_pid}_{ts_str}.csv"

    def csv_generator():
        buf = io.StringIO()
        buf.write("# HEARTIX SMART ECG TELEMETRY EXPORT\n")
        buf.write(f"# Patient Name: {patient.get('name') or 'N/A'}\n")
        buf.write(f"# Age: {patient.get('age') if patient.get('age') is not None else 'N/A'}\n")
        buf.write(f"# Gender: {patient.get('gender') or 'N/A'}\n")
        buf.write(f"# Patient ID: {patient.get('patient_id') or 'N/A'}\n")
        buf.write(f"# Bed: {patient.get('bed') or 'N/A'}\n")
        buf.write(f"# HR Limits: {patient.get('min_hr', 60)} - {patient.get('max_hr', 100)} BPM\n")
        buf.write(f"# Notes: {patient.get('notes') or 'None'}\n")
        buf.write(f"# Export Time (UTC): {now_utc.isoformat()}\n")

        writer = csv.writer(buf)
        writer.writerow(["timestamp_iso", "ecg_mv", "bpm", "status"])
        yield buf.getvalue()
        buf.seek(0)
        buf.truncate(0)

        chunk_count = 0
        for rec in records:
            ts_val = float(rec.get("ts", 0.0))
            iso_ts = datetime.fromtimestamp(ts_val, tz=timezone.utc).isoformat()
            ecg_mv = rec.get("ecg_mv", 0.0)
            bpm_val = "" if rec.get("bpm") is None else rec.get("bpm")
            status_val = rec.get("status", "")
            writer.writerow([iso_ts, ecg_mv, bpm_val, status_val])
            chunk_count += 1
            if chunk_count >= 1000:
                yield buf.getvalue()
                buf.seek(0)
                buf.truncate(0)
                chunk_count = 0

        if chunk_count > 0:
            yield buf.getvalue()

    headers = {
        "Content-Disposition": f'attachment; filename="{filename}"',
        "Cache-Control": "no-store",
    }
    return StreamingResponse(csv_generator(), media_type="text/csv", headers=headers)


@app.websocket("/ws/esp32")
async def ws_esp32_endpoint(
    websocket: WebSocket,
    token: str = Query(default=""),
) -> None:
    expected_token = os.getenv("HEARTIX_DEVICE_TOKEN", "heartix-dev-token")
    await websocket.accept()

    if not token or token != expected_token:
        await websocket.close(code=1008, reason="Invalid HEARTIX_DEVICE_TOKEN")
        return

    # Graceful Reconnect Takeover: close any stale ESP32 connection before taking over
    if hub.esp32_ws is not None and hub.esp32_ws is not websocket:
        old_ws = hub.esp32_ws
        hub.esp32_ws = None
        try:
            await old_ws.close(code=1012, reason="Superseded by new ESP32 connection")
        except Exception:
            pass

    # Hardware Priority: immediately disable simulator, reset DSP & telemetry, and notify clients
    hub.esp32_ws = websocket
    hub.esp32_connected = True
    hub.sim_mode = None
    hub.disconnected_mid_session = False
    hub.last_esp32_packet_ts = time.monotonic()
    hub.processor.reset()
    hub.reset_telemetry()
    await hub.broadcast_state()

    try:
        while True:
            message = await websocket.receive()
            msg_type = message.get("type")
            if msg_type == "websocket.disconnect":
                break

            raw_mv: list[float] = []
            leads_off: bool = False

            if message.get("bytes") is not None:
                data: bytes = message["bytes"]
                if len(data) < 4:
                    continue
                seq, flags, count = struct.unpack("<HBB", data[:4])
                leads_off = bool(flags & 0x01)
                expected_bytes = 4 + 2 * count
                if count == 0 or len(data) < expected_bytes:
                    continue
                samples_u16 = struct.unpack(f"<{count}H", data[4:expected_bytes])
                raw_mv = [float(val) for val in samples_u16]
                hub.record_esp32_packet(seq, count)

            elif message.get("text") is not None:
                try:
                    payload = json.loads(message["text"])
                except Exception:
                    continue
                if not isinstance(payload, dict):
                    continue
                leads_off = bool(payload.get("leads_off", False))
                raw_samples = payload.get("samples", [])
                if not isinstance(raw_samples, list) or len(raw_samples) == 0:
                    continue
                raw_mv = [float(val) for val in raw_samples]
                seq_val = payload.get("seq")
                hub.record_esp32_packet(
                    int(seq_val) if isinstance(seq_val, int) else None,
                    len(raw_mv),
                )
            else:
                continue

            hub.last_esp32_packet_ts = time.monotonic()
            fast_out = hub.processor.ingest_batch(raw_mv, leads_off=leads_off)

            await hub.broadcast_json(
                {
                    "type": "ecg_batch",
                    "source": "esp32",
                    "samples": fast_out["samples"],
                    "r_peak_offsets": fast_out["r_peak_offsets"],
                    "leads_off": fast_out["leads_off"],
                    "calibrating": fast_out["calibrating"],
                }
            )
    except WebSocketDisconnect:
        pass
    except Exception:
        pass
    finally:
        # Only clear state if this socket is still the active ESP32 connection
        if hub.esp32_ws is websocket:
            hub.esp32_ws = None
            hub.esp32_connected = False
            hub.disconnected_mid_session = True
            hub.processor.reset()
            hub.reset_telemetry()
            await hub.broadcast_state()
            await hub.broadcast_json(hub.build_disconnect_metrics_payload())


@app.websocket("/ws/web")
async def ws_web_endpoint(websocket: WebSocket) -> None:
    await websocket.accept()
    hub.web_clients.add(websocket)

    try:
        await websocket.send_text(json.dumps(hub.build_state_payload()))

        while True:
            raw_text = await websocket.receive_text()
            try:
                cmd = json.loads(raw_text)
            except Exception:
                continue
            if not isinstance(cmd, dict):
                continue

            action = cmd.get("action")

            if action == "simulate":
                # Hardware priority: ignore simulator commands whenever physical ESP32 is connected
                if hub.esp32_connected:
                    continue
                requested_mode = str(cmd.get("mode", "normal")).strip().lower()
                if requested_mode not in ECGSimulator.SUPPORTED_MODES:
                    continue
                if hub.sim_mode is None:
                    hub.processor.reset()
                    hub.simulator.reset()
                hub.sim_mode = requested_mode
                hub.disconnected_mid_session = False
                hub.packet_loss_pct = 0.0
                hub.effective_hz = 250.0
                await hub.broadcast_state()

            elif action == "stop_sim":
                if hub.sim_mode is not None:
                    hub.sim_mode = None
                    hub.processor.reset()
                    hub.simulator.reset()
                    hub.reset_telemetry()
                    await hub.broadcast_state()

            elif action == "pong":
                continue

    except WebSocketDisconnect:
        pass
    except Exception:
        pass
    finally:
        hub.web_clients.discard(websocket)
